PREFIX foaf: <http://xmlns.com/foaf/0.1/>

SELECT ?name ?age
WHERE
{
  ?person a foaf:Person .
  ?person foaf:name ?name .
  OPTIONAL {
      ?person foaf:age ?age .
    }
  FILTER (?age > "18"^^xsd:integer)
}
ORDER BY ?name

PREFIX foaf: <http://xmlns.com/foaf/0.1/>

SELECT ?name ?age ?email
WHERE
{
  ?p a foaf:Person .
  ?p foaf:name ?name .
  OPTIONAL {
      ?p foaf:age ?age .
    }
  OPTIONAL {
      ?p foaf:mbox ?email .
    }
  MINUS {
      ?p foaf:nick ?nick .
    }
}
ORDER BY DESC(?name)
