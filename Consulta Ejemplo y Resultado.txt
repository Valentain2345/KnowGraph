PREFIX foaf: <http://xmlns.com/foaf/0.1/>

SELECT ?name ?age
WHERE
{
  ?person a foaf:Person .
  ?person foaf:name ?name .
  OPTIONAL {
      ?person foaf:age ?age .
    }
  FILTER (?age > "18"^^xsd:integer)
}
ORDER BY ?name
